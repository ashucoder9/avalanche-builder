1. Background & Problem Statement

The DOS Chain is stuck at block 2,273,699. All 4 validators (v0, v1, v2, v6) bootstrap successfully and enter consensus, but no new blocks are produced. Debug logs confirm:

DEBUG <22v7AG7... Chain> snowman/engine.go:631 processing block not found
{"requestedHeight": 2273700, "lastAcceptedHeight": 2273699}

The root cause involves v6 not being active in the P-Chain validator set. Without v6, the remaining validators may not meet the 67% weight threshold required for consensus, which prevents block production.

Why Normal Registration Doesn't Work

Normal Flow (requires block production):
ValidatorManager.initializeValidatorRegistration() → Warp message emitted on-chain → Aggregator collects BLS signatures via P2P → RegisterL1ValidatorTx on P-Chain

Since the chain can't produce blocks, Step 1 can't execute. The initializeValidatorRegistration() transaction gets accepted into the mempool but never mined.

The Bypass: Off-Chain BLS Signing

Off-Chain Flow (no block production needed):
Construct RegisterL1ValidatorMessage manually → Sign with BLS private keys from v0, v1, v2 → Aggregate signatures off-chain → RegisterL1ValidatorTx on P-Chain

The P-Chain only checks that the Warp message has valid BLS signatures from ≥67% of the L1's validator weight. It does not verify the message originated from an on-chain transaction. We can construct and sign the message entirely off-chain using the validators' BLS private keys (signer.key files).

4.3 How It Works

The script performs these steps:

  1. Construct a RegisterL1ValidatorMessage with v6's details (NodeID, BLS public key, weight, expiry, balance/disable owners)
  2. Wrap it in an AddressedCall payload with the ValidatorManager address as the source
  3. Create an UnsignedMessage with the DOS Chain's blockchain ID and mainnet network ID
  4. Sign the unsigned message with BLS private keys from v0, v1, v2
  5. Aggregate the BLS signatures into a BitSetSignature with correct signer indices
  6. Submit RegisterL1ValidatorTx on the P-Chain with the signed warp message + v6's proof of possession + AVAX deposit

package main

import (
	"context"
	"encoding/hex"
	"fmt"
	"log"
	"sort"
	"time"

	"github.com/ava-labs/avalanchego/ids"
	"github.com/ava-labs/avalanchego/utils/constants"
	"github.com/ava-labs/avalanchego/utils/crypto/bls"
	"github.com/ava-labs/avalanchego/utils/crypto/bls/signer/localsigner"
	"github.com/ava-labs/avalanchego/utils/set"
	"github.com/ava-labs/avalanchego/utils/units"
	"github.com/ava-labs/avalanchego/vms/platformvm/warp"
	warpmessage "github.com/ava-labs/avalanchego/vms/platformvm/warp/message"
	warppayload "github.com/ava-labs/avalanchego/vms/platformvm/warp/payload"
	"github.com/ava-labs/avalanchego/wallet/subnet/primary"
)

func main() {
	ctx := context.Background()

	// ================================================================
	// CONFIGURATION — FILL IN ALL VALUES BELOW
	// ================================================================

	networkID := constants.MainnetID // 1

	blockchainID, _ := ids.FromString(
		"22v7AG7h6qaVxd4bLvAsSsg2LZ4RCn5iVYgFn7a2Fj1LCuYwjv",
	)
	subnetID, _ := ids.FromString(
		"nQCwF6V9y8VFjvMuPeQVWWYn6ba75518Dpf6ZMWZNb3NyTA94",
	)

	// ValidatorManager proxy contract address on the DOS chain
	managerAddressHex := "0x..."  // <-- FILL IN

	// v6 details (from info.getNodeID on v6)
	v6NodeIDStr      := "NodeID-Mu35k31HAXFbEp1SqJr1uUrue9nwqVFe6"
	v6BLSPubKeyHex   := "..."    // <-- FILL IN: nodePOP.publicKey
	v6BLSPoPHex      := "..."    // <-- FILL IN: nodePOP.proofOfPossession

	// Weight for v6 (match existing validators)
	v6Weight := uint64(20)       // <-- FILL IN

	// AVAX deposit for v6 on the P-Chain (10 AVAX = ~7.5 months)
	depositBalance := 10 * units.Avax

	// BLS private key files from the SIGNING validators (v0, v1, v2)
	// Copy from ~/.avalanchego/staking/signer.key on each machine
	signerKeyPaths := []string{
		"./keys/v0-signer.key",
		"./keys/v1-signer.key",
		"./keys/v2-signer.key",
	}

	// Node IDs of the signing validators (SAME ORDER as key paths)
	signerNodeIDs := []string{
		"NodeID-3K3PUAqo3cKxRoQyYto1EsXtuTHoDZ2B6",
		"NodeID-Ma3Ztm2A48bCVjfiSmoZS5MKNzKVTRN7j",
		"NodeID-G8yv8mWQy8FLdFfAhJ4v294T5Z2DWDVEH",
	}

	// P-Chain private key (hex) — needs ~11 AVAX balance
	pChainPrivKeyHex := "..."    // <-- FILL IN

	// Avalanche node RPC endpoint
	nodeURI := "http://127.0.0.1:9650"

	// P-Chain owner address (receives remaining deposit on removal)
	ownerAddressHex := "..."     // <-- FILL IN (20-byte short ID, hex)

	// ================================================================
	// END CONFIGURATION
	// ================================================================

	fmt.Println("=== DOS Chain v6 Validator Registration ===")
	fmt.Println("    Off-Chain BLS Signing Approach")
	fmt.Println()

	// --- Parse v6 details ---
	v6NodeID, err := ids.NodeIDFromString(v6NodeIDStr)
	if err != nil {
		log.Fatalf("Parse v6 NodeID: %v", err)
	}

	v6BLSPubKeyBytes, _ := hex.DecodeString(strip0x(v6BLSPubKeyHex))
	var v6BLSPubKey [bls.PublicKeyLen]byte
	copy(v6BLSPubKey[:], v6BLSPubKeyBytes)

	v6BLSPoPBytes, _ := hex.DecodeString(strip0x(v6BLSPoPHex))
	var v6PoP [bls.SignatureLen]byte
	copy(v6PoP[:], v6BLSPoPBytes)

	managerAddr, _ := hex.DecodeString(strip0x(managerAddressHex))

	ownerAddrBytes, _ := hex.DecodeString(strip0x(ownerAddressHex))
	var ownerAddr ids.ShortID
	copy(ownerAddr[:], ownerAddrBytes)

	balanceOwner := warpmessage.PChainOwner{
		Threshold: 1,
		Addresses: []ids.ShortID{ownerAddr},
	}
	disableOwner := warpmessage.PChainOwner{
		Threshold: 1,
		Addresses: []ids.ShortID{ownerAddr},
	}

	expiry := uint64(time.Now().Add(24 * time.Hour).Unix())

	// ================================================================
	// STEP 1: Construct RegisterL1Validator message
	// ================================================================
	fmt.Println("[1/5] Constructing RegisterL1Validator message...")

	registerMsg, err := warpmessage.NewRegisterL1Validator(
		subnetID, v6NodeID, v6BLSPubKey,
		expiry, balanceOwner, disableOwner, v6Weight,
	)
	if err != nil {
		log.Fatalf("Create message: %v", err)
	}

	validationID := registerMsg.ValidationID()
	fmt.Printf("      ValidationID: %s\n", validationID)
	fmt.Printf("      Expiry:       %s\n",
		time.Unix(int64(expiry), 0).UTC().Format(time.RFC3339))

	// Wrap in AddressedCall
	addrCall, err := warppayload.NewAddressedCall(
		managerAddr, registerMsg.Bytes(),
	)
	if err != nil {
		log.Fatalf("Create AddressedCall: %v", err)
	}

	// Create unsigned warp message
	unsignedWarp, err := warp.NewUnsignedMessage(
		networkID, blockchainID, addrCall.Bytes(),
	)
	if err != nil {
		log.Fatalf("Create unsigned warp: %v", err)
	}

	fmt.Printf("      Warp msg hash: %s\n", unsignedWarp.ID())

	// ================================================================
	// STEP 2: Determine canonical validator ordering
	// ================================================================
	fmt.Println("\n[2/5] Determining canonical validator ordering...")

	var allNodeIDs []ids.NodeID
	for _, nidStr := range signerNodeIDs {
		nid, _ := ids.NodeIDFromString(nidStr)
		allNodeIDs = append(allNodeIDs, nid)
	}

	// Sort by NodeID bytes (canonical P-Chain ordering)
	sort.Slice(allNodeIDs, func(i, j int) bool {
		return allNodeIDs[i].Compare(allNodeIDs[j]) < 0
	})

	nodeIDToIndex := make(map[ids.NodeID]int)
	for i, nid := range allNodeIDs {
		fmt.Printf("      [%d] %s\n", i, nid)
		nodeIDToIndex[nid] = i
	}

	// ================================================================
	// STEP 3: Load BLS keys and sign the warp message
	// ================================================================
	fmt.Println("\n[3/5] Loading BLS keys and signing...")

	signerBits := set.NewBits()
	var blsSigs []*bls.Signature

	for i, keyPath := range signerKeyPaths {
		sk, err := localsigner.FromFile(keyPath)
		if err != nil {
			log.Fatalf("Load key %s: %v", keyPath, err)
		}

		nid, _ := ids.NodeIDFromString(signerNodeIDs[i])
		idx := nodeIDToIndex[nid]

		sig, err := sk.Sign(unsignedWarp.Bytes())
		if err != nil {
			log.Fatalf("Sign with %s: %v", keyPath, err)
		}

		signerBits.Add(idx)
		blsSigs = append(blsSigs, sig)
		fmt.Printf("      Signed: %s (index %d)\n", signerNodeIDs[i], idx)
	}

	// ================================================================
	// STEP 4: Aggregate signatures
	// ================================================================
	fmt.Println("\n[4/5] Aggregating BLS signatures...")

	aggSig, err := bls.AggregateSignatures(blsSigs)
	if err != nil {
		log.Fatalf("Aggregate: %v", err)
	}

	var aggSigBytes [bls.SignatureLen]byte
	copy(aggSigBytes[:], bls.SignatureToBytes(aggSig))

	signedWarp, err := warp.NewMessage(
		unsignedWarp,
		&warp.BitSetSignature{
			Signers:   signerBits.Bytes(),
			Signature: aggSigBytes,
		},
	)
	if err != nil {
		log.Fatalf("Create signed warp: %v", err)
	}

	fmt.Printf("      Signed message: %d bytes\n", len(signedWarp.Bytes()))

	// ================================================================
	// STEP 5: Submit RegisterL1ValidatorTx on P-Chain
	// ================================================================
	fmt.Println("\n[5/5] Submitting RegisterL1ValidatorTx on P-Chain...")

	pKeyBytes, _ := hex.DecodeString(strip0x(pChainPrivKeyHex))
	kc := primary.NewKeychain(pKeyBytes...)
	wallet, err := primary.MakePWallet(
		ctx, nodeURI, kc,
		primary.WalletConfig{SubnetIDs: []ids.ID{subnetID}},
	)
	if err != nil {
		log.Fatalf("Create wallet: %v", err)
	}

	tx, err := wallet.IssueRegisterL1ValidatorTx(
		depositBalance, v6PoP, signedWarp.Bytes(),
	)
	if err != nil {
		log.Fatalf("Issue tx: %v", err)
	}

	fmt.Println("\n========================================")
	fmt.Println("  SUCCESS")
	fmt.Println("========================================")
	fmt.Printf("  TX ID:         %s\n", tx.ID())
	fmt.Printf("  ValidationID:  %s\n", validationID)
	fmt.Printf("  Deposit:       %.2f AVAX\n",
		float64(depositBalance)/float64(units.Avax))
	fmt.Println("\n  v6 is now registered on the P-Chain.")
	fmt.Println("  Proceed to Section 6: Warp Timestamp Fix.")
}

func strip0x(s string) string {
	if len(s) >= 2 && s[:2] == "0x" {
		return s[2:]
	}
	return s
}

Appendix B: Security Considerations

  • BLS Private Keys (signer.key): These are the most sensitive artifacts. They prove validator identity and can sign arbitrary warp messages. Transfer only via SSH/SCP. Delete all copies from the working machine immediately after the registration transaction is confirmed.
  • P-Chain Private Key: The hex private key used for submitting the P-Chain transaction should be treated as a hot wallet key. Only load the minimum AVAX needed (~11 AVAX).
  • Do NOT commit keys to git. Add keys/ to .gitignore before starting.
  • Expiry: The registration message has a 24-hour expiry. If the transaction is not submitted in time, the message becomes invalid. Re-run the script to generate a new one.
  • Weight: Setting an incorrect weight can affect consensus thresholds. Match existing validator weights unless intentionally rebalancing.
